cmake_minimum_required(VERSION 3.16)
project(smartdrive_rgbd_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rgbd_geometry STATIC src/rgbd_geometry.cpp)
target_include_directories(rgbd_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(test_rgbd_geometry tests/test_geometry.cpp)
target_link_libraries(test_rgbd_geometry PRIVATE rgbd_geometry)
add_executable(rgbd_geometry_cli src/geometry_cli.cpp)
target_link_libraries(rgbd_geometry_cli PRIVATE rgbd_geometry)
enable_testing()
add_test(NAME rgbd_geometry_unit COMMAND test_rgbd_geometry)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME python_cpp_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_python_parity.py $<TARGET_FILE:rgbd_geometry_cli>)
endif()

add_executable(geometry_benchmark src/geometry_benchmark.cpp)
target_link_libraries(geometry_benchmark PRIVATE rgbd_geometry)
